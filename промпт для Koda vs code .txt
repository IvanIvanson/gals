            const canvas = document.getElementById("canvas");

            const ctx = canvas.getContext("2d");

            const cellSizeInput = document.getElementById("cellSize");

            const sizeValue = document.getElementById("sizeValue");

            const showVerticalCheckbox = document.getElementById("showVertical");

            

            let cellSize = 30; // Начальный размер ячейки

            let showVertical = true; // Показывать вертикальные линии

            

            // Обработчики событий

            cellSizeInput.addEventListener("input", updateGrid);

            showVerticalCheckbox.addEventListener("change", updateGrid);

            

            function updateGrid() {

                cellSize = parseInt(cellSizeInput.value);

                sizeValue.textContent = cellSize;

                showVertical = showVerticalCheckbox.checked;

                drawGrid();

            }

            

            function drawGrid() {

                ctx.clearRect(0, 0, canvas.width, canvas.height);

                ctx.translate(0.5, 0.5); // Для чёткости линий

                

                // Горизонтальные линии

                const horizontalLines = Math.floor(canvas.height / cellSize);

                for(let i = 0; i <= horizontalLines; i++) {

                    const y = i * cellSize;

                    ctx.lineWidth = (i === 5) ? 3 : 1; // 5-я линия толще

                    ctx.strokeStyle = (i % 2 === 0) ? "red" : "green";

                    ctx.beginPath();

                    ctx.moveTo(0, y);

                    ctx.lineTo(canvas.width, y);

                    ctx.stroke();

                }

                

                // Вертикальные линии (если включены)

                if(showVertical) {

                    const verticalLines = Math.floor(canvas.width / cellSize);

                    for(let i = 0; i <= verticalLines; i++) {

                        const x = i * cellSize;

                        ctx.lineWidth = (i === 5) ? 3 : 1; // 5-я линия толще

                        ctx.strokeStyle = (i % 2 === 0) ? "red" : "green";

                        ctx.beginPath();

                        ctx.moveTo(x, 0);

                        ctx.lineTo(x, canvas.height);

                        ctx.stroke();

                    }

                }

                

                ctx.translate(-0.5, -0.5); // Возвращаем смещение

            }

            

            // Инициализация

            updateGrid();

        </script>
