Verification form

Stable

Код подтверждения из четырёх ячеек: цифры распределяются по полям, Backspace и стрелки переносят фокус, вставка разкладывается по всем ячейкам сразу.

Usage

<div class="verify-code">
  <h2 id="vc-title">Validation code</h2>
  <p>We have sent a verification code to your email.</p>

  <form onsubmit="return confirmCode(this)">
    <div class="code-fields" role="group" aria-labelledby="vc-title">
      <input class="code-input" type="text" inputmode="numeric"
             autocomplete="one-time-code" pattern="[0-9]" maxlength="1" required
             aria-label="Digit 1 of 4"
             oninput="codeInput(this)"
             onkeydown="codeKeydown(this, event)"
             onpaste="codePaste(this, event)">
      <input class="code-input" type="text" inputmode="numeric"
             autocomplete="off" pattern="[0-9]" maxlength="1" required
             aria-label="Digit 2 of 4"
             oninput="codeInput(this)"
             onkeydown="codeKeydown(this, event)"
             onpaste="codePaste(this, event)">
      <!-- …ещё две ячейки -->
    </div>

    <!-- Собранный код уходит на сервер одним полем -->
    <input type="hidden" name="code">
    <button class="btn btn-primary" type="submit">Confirm</button>
    <p class="code-status" role="status" aria-live="polite"></p>
  </form>
</div>

.verify-code { width: 100%; max-width: 323px; }
.code-fields {

  gap: 10px; margin: 0 0 20px;

.code-input {

  min-height: 44px; aspect-ratio: 1; padding: 0;
  font-size: 20px; text-align: center;
  border: 2px solid #ddd; border-radius: 15px;
}
.code-input.is-filled { border-color: #bd5518; }
.code-input:focus-visible {
  outline: 3px solid #6b2705; outline-offset: 2px;
  border-color: #a8410c;
}
.code-status { min-height: 20px; margin: 12px 0 0; font-size: 14px; }
.code-status.is-ok  { color: #16692c; }
.code-status.is-err { color: #b3261e; }
</style>

<script>
function codeInput(input) {
  var fields = Array.prototype.slice.call(
    input.closest('.verify-code').querySelectorAll('.code-input'));
  var index = fields.indexOf(input);

  // maxlength="1" не спасает от автоподстановки кода из SMS: цифр может
  // прийти несколько — раздаём их по следующим полям.
  var digits = input.value.replace(/D/g, '');
  input.value = digits.slice(0, 1);

  var cursor = index;
  for (var i = 1; i < digits.length && cursor + 1 < fields.length; i++) {
    fields[++cursor].value = digits.charAt(i);
  }

  fields.forEach(function (f) {
    f.classList.toggle('is-filled', f.value !== '');
  });
  if (input.value && cursor + 1 < fields.length) fields[cursor + 1].focus();
}

function codeKeydown(input, event) {
  var fields = Array.prototype.slice.call(
    input.closest('.verify-code').querySelectorAll('.code-input'));
  var index = fields.indexOf(input);
  var last = fields.length - 1;

  if (event.key === 'Backspace' && input.value === '' && index > 0) {
    fields[index - 1].focus();
  } else if (event.key === 'ArrowLeft' && index > 0) {
    fields[index - 1].focus();
  } else if (event.key === 'ArrowRight' && index < last) {
    fields[index + 1].focus();
  }
}

function codePaste(input, event) {
  var fields = Array.prototype.slice.call(
    input.closest('.verify-code').querySelectorAll('.code-input'));
  var index = fields.indexOf(input);
  var digits = event.clipboardData.getData('text').replace(/D/g, '');
  if (!digits) return;

  event.preventDefault();
  // Вставка идёт от активного поля, а не всегда с первого.
  fields.forEach(function (f, i) {
    if (i >= index) f.value = digits.charAt(i - index) || '';
  });
  fields.forEach(function (f) {
    f.classList.toggle('is-filled', f.value !== '');
  });

  var empty = fields.slice(index).filter(function (f) { return f.value === ''; })[0];
  (empty || fields[fields.length - 1]).focus();
}

function confirmCode(form) {
  var fields = Array.prototype.slice.call(
    form.querySelectorAll('.code-input'));
  var code = fields.map(function (f) { return f.value; }).join('');
  var status = form.querySelector('.code-status');
  form.querySelector('[name="code"]').value = code;

  if (code.length !== fields.length) {
    status.className = 'code-status is-err';
    status.textContent = 'Enter all four digits.';
    fields.filter(function (f) { return f.value === ''; })[0].focus();
    return false;
  }

  status.className = 'code-status is-ok';
  status.textContent = 'Code ' + code + ' confirmed.';
  return false;
}
</script>

Notes

Главный баг исходника: index < index.length - 1. У числа нет свойства length, сравнение 0 < undefined всегда false — стрелка → не работала никогда. Исправлено на index < fields.length - 1.

document.querySelectorAll('input') выбирал все input страницы — включая поиск в сайдбаре. Заменено на root.querySelectorAll('.demo-vf-input').

window.onload в документации не сработал бы: HTML вставляется через innerHTML, и скрипты внутри не исполняются. Обработчики вынесены в window.__galsComponents и вызываются из inline-атрибутов.

Вставка игнорировала активное поле: код всегда ложился в inputs[0..], поэтому вставка из третьей ячейки перезаписывала первые. Теперь вставка идёт от активного поля.

Автоподстановка кода из SMS ломалась: maxlength="1" плюс .charAt(0) выбрасывали лишние цифры, которые присылает браузер. Теперь пришедшие цифры раздаются по следующим полям. Первому полю добавлен autocomplete="one-time-code" — именно он включает системную подсказку кода на iOS, Android и в Chrome.

Класс .active разъезжался: paste вешал его на все поля, Backspace снимал только с одного, а правка первой ячейки оставляла подсветку на остальных. Теперь состояние всех ячеек пересчитывается целиком (paintCode).

outline: none без замены лишал поле видимого фокуса — клавиатурная навигация по коду становилась вслепую. Добавлено кольцо :focus-visible.

Четыре голых input скринридер объявляет как «edit text» без объяснения. Добавлены aria-label="Digit N of 4" и role="group" с aria-labelledby на контейнер.

Отправлялся maxlength="1" без name — на сервер не уходило ничего. Добавлено скрытое поле name="code", куда хелпер складывает собранный код.

Форма вела на action="#" и ничего не сообщала о результате. Добавлен <p role="status" aria-live="polite"> — он объявляет и ошибку, и успех. Здесь он подтверждает любой код: настоящая проверка живёт на сервере, клиент не может ей доверять.

Глобальные * { margin:0; padding:0 }, body, h2, p, input, button и form > div обнулили бы отступы и перекрасили формы всей документации. Все селекторы скоуплены к .demo-vf-*.

Фиксированная height: 350px блоки растягивал текст, масштаб или перевод. Заменена на автовысоту; padding: 90px 38px 38px — на обычный отступ.

Подпись #b6b6b6 на белом — контраст ~2:1, код не читается. Тёмно-оранжевая кнопка с белым текстом (#ff8d4d) давала ~2.3:1. Оба цвета затемнены до ~5:1 и ~4.7:1.

Мелочи: padding:4 без единиц CSS игнорирует (исправлено на 0), <button> без type внутри формы отправлял бы её молча — теперь type="submit" явно, а onsubmit возвращает false.
